```python
# Creacion de ficheros
BASE = '/datalake/renfe'
ENC = 'utf-8'          

for d in [f'{BASE}/raw/estaciones', f'{BASE}/raw/viajeros',
          f'{BASE}/processed', f'{BASE}/logs']:
    client.makedirs(d, permission='755')

```


```python
# Subida
uploads = [
    ('estaciones_completo.csv', f'{BASE}/raw/estaciones/estaciones_completo.csv'),
    ('estaciones_feve.csv',     f'{BASE}/raw/estaciones/estaciones_feve.csv'),
    ('viajeros_tramo.csv',      f'{BASE}/raw/viajeros/viajeros_tramo.csv'),
]
for local, remoto in uploads:
    client.upload(remoto, local, n_threads=4, overwrite=True)
```


```python
# Auditoria
```

```python
# Filtro

filtradas = []
contador = 0

with client.read(f'{BASE}/raw/estaciones/estaciones_completo.csv', encoding=ENC, delimiter='\n') as reader:
    for i, linea in enumerate(reader):
        if i == 0:
            filtradas.append(linea)
        elif "LEÓN" in linea.upper():
            filtradas.append(linea)
            contador += 1

client.write(f'{BASE}/processed/estaciones_filtradas.csv', data='\n'.join(filtradas) + '\n', overwrite=True, encoding=ENC)
print(f'Estaciones filtradas: {contador}')
```
